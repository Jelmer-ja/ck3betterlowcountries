ellert_cycle.1001 = {
	type = character_event
	title = ellert_cycle.1001.t
	desc = ellert_cycle.1001.desc

	trigger = {
		has_ep3_dlc_trigger = yes
	}

	immediate = {
		create_character = {
			name = "David"
			culture = culture:saxon
			faith = faith:catholic
			location = province:2417
			prowess = 15
			gender_female_chance = 0
			save_scope_as = spawned_peasant_1
		}
		root.location = { save_scope_as = terrain_scope }
	}

	theme = learning
	override_background = { reference = temple }
	left_portrait = {
		character = root
		animation = personality_honorable
	}
	right_portrait = {
		character = scope:spawned_peasant_1
		animation = anger
	}

	option = {
		name = ellert_cycle.1001.a
		ai_chance = {
			base = 1000
		}

        configure_start_single_combat_effect = {
            SC_INITIATOR = ROOT
            SC_ATTACKER = scope:spawned_peasant_1
            SC_DEFENDER = ROOT

			LOCALE = terrain_scope

            FATALITY = always
            FIXED = no

            OUTPUT_EVENT = ellert_cycle.1004
            INVALIDATION_EVENT = ellert_cycle.1004
        }
	}
}
