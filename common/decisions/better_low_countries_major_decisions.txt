form_netherlands_kingdom_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_unite_netherlands.dds"
	}
	desc = form_netherlands_kingdom_decision_desc
	selection_tooltip = form_netherlands_kingdom_decision_tooltip
	decision_group_type = major

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 120
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	is_shown = {
		NOT = { exists = title:k_netherlands.holder }
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_formed_netherlands_kingdom
			}
		}
		#No kings or emperors.
		highest_held_title_tier <= tier_duchy
		#Must be within an empire.
		OR = {
			culture = { has_cultural_pillar = heritage_central_germanic }
			culture = { has_cultural_pillar = heritage_north_germanic } # for vikings in 876 bookmark
			culture = { has_cultural_pillar = heritage_frankish }
		}
		any_sub_realm_county = {
			title_province = { geographical_region = custom_core_netherlands_full }
		}
	}

	is_valid = {
		prestige_level >= high_prestige_level
		#And at least Holland, Brabant, Gelre, Julich, Flanders, Utrecht and Luxemburg.
		completely_controls_region = custom_core_netherlands
		custom_tooltip = {
			text = liege_is_emperor_or_independent_requirement_tt
			OR = {
				top_liege = this
				top_liege.primary_title.tier >= tier_empire
			}
		}
	}

	is_valid_showing_failures_only = {
		is_landed = yes
		is_capable_adult = yes
		is_imprisoned = no
		is_at_war = no
	}

	cost = {
		gold = {
			value = 0
			if = {
				limit = {
					has_treasury = no
				}
				add = 500
			}
		}
		treasury = {
			value = 0
			if = {
				limit = {
					has_treasury = yes
				}
				add = 500
			}
		}
		prestige = {
			value = major_prestige_gain
			multiply = 3
			round = yes
		}
		piety = {
			value = major_piety_gain
			multiply = 3
			round = yes
		}
	}

	effect = {
		gain_heroic_legend_seed_tooltip_effect = yes
		show_as_tooltip = {
			form_netherlands_scripted_effect = yes
		}
		trigger_event = {
			id = better_low_countries_decisions.0001
		}
		hidden_effect = {
			add_to_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_formed_netherlands_kingdom
			}
			set_global_variable = {
				name = flag_formed_netherlands_kingdom
				value = root
			}
		}
	}
	
	ai_potential = {
		always = yes
	}

	ai_will_do = {
		base = 100
	}
}

	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_found_kingdom.dds"
	}
	title = {
		first_valid = {
			triggered_desc = {

					primary_title = title:k_frisia
				}
				desc = merge_frisia_decision_alt
			}
			desc = merge_frisia_decision
		}
	}
	desc = {
		first_valid = {
			triggered_desc = {
				trigger = {
					primary_title = title:k_frisia
				}
				desc = merge_frisia_decision_desc_frisia
			}
			desc = merge_frisia_decision_desc
		}
		desc = merge_frisia_decision_desc_outro
	}
	decision_group_type = major
	sort_order = 50

	is_shown = {
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_merged_frisia
			}
		}
		OR = {
			culture = { has_cultural_pillar = heritage_frankish }
			culture = { has_cultural_pillar = heritage_central_germanic }
		}
		OR = {
			has_title = title:k_frisia
			has_title = title:k_lotharingia
		}
		NOR = {
			title:k_frisia = { is_titular = yes }
			title:k_lotharingia = { is_titular = yes }
		}
	}

	is_valid = {
		has_title = title:k_frisia
		has_title = title:k_lotharingia
		top_liege = this
		prestige_level >= 3
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_merged_frisia
		}
		set_global_variable = {
			name = flag_restored_pagan_saxony
			value = root
		}

		save_scope_as = rightful_liege

		add_legitimacy_effect = { LEGITIMACY = massive_legitimacy_gain }
		dynasty ?= { add_dynasty_prestige = massive_dynasty_prestige_value }

		if = {
			limit = {
				primary_title = { this = title:k_frisia}

				PRIMARY = k_frisia

				TEXT = alt
			}
		}
		else = {
			merge_frisia_effect = {
				PRIMARY = k_france
				SECONDARY = k_frisia
				TEXT = original
			}
		}

		trigger_event = better_low_countries_decisions.0007
		every_vassal_or_below = {
			trigger_event = better_low_countries_decisions.0007
		}
	}

	cost = {
		prestige = {
			if = {
				limit = {
					is_ai = no
				}
				value = 5000
			}
			else = {

			}
		}
	}

ubbe_abandon_walcheren_decision = {
	picture = {
		reference = "gfx/interface/illustrations/decisions/fp1_decision_forge_jomsvikings.dds"
	}
	decision_group_type = major
	sort_order = 82
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 120
		duchy = 120
		kingdom = 0
		empire = 0
		hegemony = 0
	}

	desc = ubbe_abandon_walcheren_decision_desc
	selection_tooltip = ubbe_abandon_walcheren_decision_tooltip

	is_shown = {
		has_ep3_dlc_trigger = yes
		is_ai = no #AI Ubbe will never abandon his lands
		this = character:163119
		is_landed = yes
		domain_size <= 3
		NOT = {  #Can only do it once.
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:flag_ubbe_left
			}
		}

		# Relevant cultures/faiths
		save_temporary_scope_as = potential_laamp_temp
		can_be_combatant_based_on_gender_trigger = { ARMY_OWNER = scope:potential_laamp_temp } # Registered fighting license
	}

	is_valid = {
		is_valid_for_laampdom = yes
		# Is not HOF
		custom_tooltip = {
			text = become_landless_adventurer_decision_hof_tt
			faith.religious_head_title.holder != root
		}
		# Counts and dukes only
		custom_tooltip = {
			text = become_landless_adventurer_decision_tier_tt
			highest_held_title_tier >= tier_county
			highest_held_title_tier <= tier_duchy
		}
		# Not too big
		custom_tooltip = {
			text = become_landless_adventurer_decision_domain_size_tt
			domain_size <= 3
		}
	}

	is_valid_showing_failures_only = {
		is_alive = yes
		is_available_adult = yes
		has_contagious_deadly_disease_trigger = no
	}

	effect = {
		save_scope_as = adventurer
		if = {
			limit = {
				player_heir ?= {
					this != root
					is_adult = yes
				}
			}
			player_heir = {
				save_scope_as = laamp_heir
				save_scope_as = laamp_inheritor
				custom_tooltip = become_landless_adventurer_decision_title_scope_tt
			}
		}
		else_if = {
			limit = { top_liege != this }
			liege = {
				save_scope_as = laamp_liege
				save_scope_as = laamp_inheritor
				custom_tooltip = become_landless_adventurer_decision_title_scope_tt
			}
		}
		else = {
			custom_tooltip = become_landless_adventurer_decision_title_created_tt
			if = {
				limit = {
					NOT = { exists = scope:laamp_inheritor }
				}
				hidden_effect_new_object = {
					create_character = {
						template = laamp_inheritor_template
						culture = root.capital_county.culture
						faith = root.capital_county.faith
						dynasty = generate
						location = root.capital_province
						after_creation = {
							save_scope_as = laamp_inheritor
							add_character_flag = new_laamp_inheritor
						}
					}
				}
			}
		}
		create_landless_adventurer_title_tooltip_effect = yes
		show_as_tooltip = {
			add_prestige = medium_prestige_gain
			add_character_modifier = {
				modifier = ep3_voluntary_laamp_character_modifier
				years = 5
			}
		}
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:flag_ubbe_left
		}
		set_global_variable = {
			name = flag_ubbe_left
			value = true
		}
		trigger_event = { id = better_low_countries_decisions.0008 }
	}

	ai_potential = {
		is_valid_for_laampdom = yes
		trigger_if = { # Temujun stays, he needs to become Genghis Khan after all!
			limit = {
				exists = character:125501
			}
			NOT = { this = character:125501 }
		}
	}

	ai_will_do = {
		base = -75
		modifier = { add = ep3_laamp_chance_score_value }
		modifier = { # If we don't have this, everyone will leave your realm constantly
			add = -45
			culture = { has_cultural_parameter = much_more_likely_to_be_laamps }
		}
		modifier = { # If we don't have this, everyone will leave your realm constantly
			add = -20
			culture = { has_cultural_parameter = more_likely_to_be_laamps }
		}
		modifier = {
			factor = 0
			OR = {
				#Stahp it, John de Courcy
				this = character:214000
				#No adventurer should want to be one again... right?
				has_variable = adventurer_creation_reason
			}
		}
	}
}
